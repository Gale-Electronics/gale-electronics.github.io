{%- comment -%}
Plain-text twin of the jekyll-sitemap /sitemap.xml, served at a second URL.
Google Search Console can stay stuck on "Couldn't fetch" for a GitHub Pages
/sitemap.xml (github.com/orgs/community/discussions/199426); submitting this
file instead gets the URLs read. Same rules as jekyll-sitemap: every page,
document and PDF unless it sets sitemap: false.
{%- endcomment -%}
{%- capture urls -%}
{%- for collection in site.collections -%}
{%- if collection.output -%}
{%- for doc in collection.docs -%}
{%- unless doc.sitemap == false %}
{{ doc.url | replace: '/index.html', '/' | absolute_url }}
{%- endunless -%}
{%- endfor -%}
{%- endif -%}
{%- endfor -%}
{%- for page in site.html_pages -%}
{%- unless page.sitemap == false or page.url == '/404.html' %}
{{ page.url | replace: '/index.html', '/' | absolute_url }}
{%- endunless -%}
{%- endfor -%}
{%- for file in site.static_files -%}
{%- if file.extname == '.pdf' -%}
{%- unless file.sitemap == false %}
{{ file.path | absolute_url }}
{%- endunless -%}
{%- endif -%}
{%- endfor %}
{%- endcapture -%}
{{ urls | strip }}
